cmake_minimum_required(VERSION 3.20)
project(AppGPUEcoSim LANGUAGES CXX)

# ── C++ standard ───────────────────────────────────────────────────────────────
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(APP_DIR "${CMAKE_SOURCE_DIR}/AppGPUEcoSim")
set(SRC_DIR "${APP_DIR}/Source")
set(INC_DIR "${APP_DIR}/Include")
set(UI_DIR  "${APP_DIR}/UI")

# ── Dependencies ───────────────────────────────────────────────────────────────
# Reuse the QTDIR64 environment variable set in Visual Studio / system settings.
if(DEFINED ENV{QTDIR64})
    list(PREPEND CMAKE_PREFIX_PATH "$ENV{QTDIR64}")
endif()

find_package(Qt6 REQUIRED COMPONENTS
    Core Gui Widgets OpenGL OpenGLWidgets PrintSupport Svg Sql)
find_package(OpenGL REQUIRED)

# GLEW: vendored under AppGPUEcoSim/ThirdParty/GLEW on Windows, so no external
# env var or shared lib tree is needed there. On Linux, GLEW is normally
# installed system-wide, so find_package works without hints.
if(WIN32)
    set(GLEW_VENDOR_DIR "${APP_DIR}/ThirdParty/GLEW")
    if(NOT EXISTS "${GLEW_VENDOR_DIR}/Library/glew32.lib")
        message(FATAL_ERROR "Vendored GLEW not found at ${GLEW_VENDOR_DIR}")
    endif()
    add_library(GLEW::GLEW UNKNOWN IMPORTED)
    set_target_properties(GLEW::GLEW PROPERTIES
        IMPORTED_LOCATION
            "${GLEW_VENDOR_DIR}/Library/glew32.lib"
        INTERFACE_INCLUDE_DIRECTORIES
            "${GLEW_VENDOR_DIR}/Include"
    )
else()
    find_package(GLEW REQUIRED)
endif()

find_package(OpenMP)

# ── Qt code generation ─────────────────────────────────────────────────────────
# Mirrors the custom build steps in the .vcxproj (preserving original filenames).

# UIC: UI/main.ui → Include/uic_main.h
add_custom_command(
    OUTPUT  "${INC_DIR}/uic_main.h"
    COMMAND Qt6::uic "${UI_DIR}/main.ui" -o "${INC_DIR}/uic_main.h"
    DEPENDS "${UI_DIR}/main.ui"
    COMMENT "UIC main.ui → uic_main.h"
    VERBATIM
)

# MOC: Include/qte.h → Source/moc_qte.cpp
add_custom_command(
    OUTPUT  "${SRC_DIR}/moc_qte.cpp"
    COMMAND Qt6::moc "${INC_DIR}/qte.h" -o "${SRC_DIR}/moc_qte.cpp"
            -I "${INC_DIR}"
    DEPENDS "${INC_DIR}/qte.h"
    COMMENT "MOC qte.h → moc_qte.cpp"
    VERBATIM
)

# MOC: Include/realtime.h → Source/realtime-moc.cpp
add_custom_command(
    OUTPUT  "${SRC_DIR}/realtime-moc.cpp"
    COMMAND Qt6::moc
            -frealtime.h
            "${INC_DIR}/realtime.h"
            -o "${SRC_DIR}/realtime-moc.cpp"
            -I "${INC_DIR}"
    DEPENDS "${INC_DIR}/realtime.h"
    COMMENT "MOC realtime.h → realtime-moc.cpp"
    VERBATIM
)

# Binary resources — loaded at runtime via QResource::registerResource,
# not embedded in the binary.  Output matches the original vcxproj path.
qt6_add_binary_resources(app_resources
    "${UI_DIR}/resources.qrc"
    DESTINATION "${UI_DIR}/resources.rcc"
)

# ── Executable ─────────────────────────────────────────────────────────────────
# WIN32 suppresses the console window on Windows; ignored on other platforms.
add_executable(AppGPUEcoSim WIN32 ${SOURCES})
add_dependencies(AppGPUEcoSim app_resources)
# qt6_add_binary_resources creates <build>/AppGPUEcoSim/UI/, which collides with
# the executable name, so give the binary a different file name.
set_target_properties(AppGPUEcoSim PROPERTIES OUTPUT_NAME "EcoSim")

# ── Runtime library search path (Linux) ───────────────────────────────────────
# Qt lives in a non-system prefix here (QTDIR64), so bake its lib dir into the
# binary's RPATH — otherwise the dynamic linker can't find libQt6*.so at
# launch (Qt6Sql included). Qt's own plugin lookup (sqldrivers/libqsqlite.so
# etc.) is relative to that lib dir too, so it resolves for free once RPATH
# is set. GLEW is a system package on Linux, so it needs no RPATH entry.
if(UNIX)
    set(_rpath_dirs "")
    if(DEFINED ENV{QTDIR64})
        list(APPEND _rpath_dirs "$ENV{QTDIR64}/lib")
    endif()
    set_target_properties(AppGPUEcoSim PROPERTIES
        BUILD_RPATH   "${_rpath_dirs}"
        INSTALL_RPATH "${_rpath_dirs}"
    )
endif()

# ── Include paths ──────────────────────────────────────────────────────────────
target_include_directories(AppGPUEcoSim PRIVATE "${INC_DIR}")

# ── Compile definitions ────────────────────────────────────────────────────────
# SOLUTION_DIR is used at runtime to locate shaders and data files relative to
# the repo root.  CMAKE_SOURCE_DIR always uses forward slashes, which Windows
# accepts in path APIs too, so no raw-string workaround is needed.
target_compile_definitions(AppGPUEcoSim PRIVATE
    "SOLUTION_DIR=\"${CMAKE_SOURCE_DIR}/\""
)

if(MSVC)
    target_compile_definitions(AppGPUEcoSim PRIVATE _MBCS)
endif()

# ── Compiler flags ─────────────────────────────────────────────────────────────
if(MSVC)
    target_compile_options(AppGPUEcoSim PRIVATE
        /Zc:__cplusplus   # expose correct __cplusplus value
        /permissive-      # strict standard conformance
        /MP               # parallel compilation
        /W3
    )
else()
    target_compile_options(AppGPUEcoSim PRIVATE
        -Wall -Wextra
        # suppress noisy warnings common in this kind of math-heavy codebase
        -Wno-unused-parameter
        -Wno-sign-compare
    )
endif()

# GLU: available as a system lib on Windows; optional on Linux
if(OpenGL_GLU_FOUND)
    target_link_libraries(AppGPUEcoSim PRIVATE OpenGL::GLU)
elseif(WIN32)
    target_link_libraries(AppGPUEcoSim PRIVATE glu32)
endif()

# OpenMP
if(OpenMP_CXX_FOUND)
    target_link_libraries(AppGPUEcoSim PRIVATE OpenMP::OpenMP_CXX)
endif()
